   256→   256→  render: function(delta){
   257→   257→    var ctx = this.ctx, W = this.W, H = this.H;
   258→   258→    this.time += delta;
   259→   259→    ctx.save();
   260→   260→    // ---- screenshake por trauma (offset + micro-rotação) ----
   261→   261→    if (this.shake > 0.001) {
   262→   262→      var tr2 = this.shake * this.shake;
   263→   263→      var sx = (Math.random() * 2 - 1) * tr2 * 11;
   264→   264→      var sy = (Math.random() * 2 - 1) * tr2 * 9;
   265→   265→      var rot = Math.sin(this.time * 57) * tr2 * 0.013;
   266→   266→      ctx.translate(W / 2 + sx, H / 2 + sy);
   267→   267→      ctx.rotate(rot);
   268→   268→      ctx.translate(-W / 2, -H / 2);
   269→   269→      this.shake = Math.max(0, this.shake - delta * 1.7);
   270→   270→    }
   271→   271→    ctx.clearRect(-30, -30, W + 60, H + 60);
   272→   272→    // ---- 1) céu ----
   273→   273→    this._drawLayers(delta);
   274→   274→    // ---- 2) névoa oscilante ----
   275→   275→    if (this.pal) {
   276→   276→      var fogY1 = this.GROUND_Y - 130 + Math.sin(this.time * 0.5) * 12;
   277→   277→      var fogY2 = this.GROUND_Y - 60 + Math.cos(this.time * 0.4) * 10;
   278→   278→      ctx.globalAlpha = 0.07;
   279→   279→      ctx.fillStyle = this.pal.accent;
   280→   280→      ctx.fillRect(0, fogY1, W, 70);
   281→   281→      ctx.globalAlpha = 0.05;
   282→   282→      ctx.fillRect(0, fogY2, W, 90);
   283→   283→      ctx.globalAlpha = 1;
   284→   284→    }
   285→   285→    // ---- 3) partículas atmosféricas (atrás das entidades) ----
   286→   286→    this._drawAtmo(delta);
   287→   287→    // ---- 4) chão ----
   288→   288→    if (this.pal) {
   289→   289→      var g = ctx.createLinearGradient(0, this.GROUND_Y, 0, H);
   290→   290→      g.addColorStop(0, this.pal.ground);
   291→   291→      g.addColorStop(1, '#040208');
   292→   292→      ctx.fillStyle = g;
   293→   293→      ctx.fillRect(0, this.GROUND_Y, W, H - this.GROUND_Y);
   294→   294→      ctx.fillStyle = 'rgba(255,255,255,0.05)';
   295→   295→      ctx.fillRect(0, this.GROUND_Y, W, 2);
   296→   296→      ctx.globalAlpha = 0.10; ctx.fillStyle = this.pal.accent;
   297→   297→      ctx.fillRect(0, this.GROUND_Y, W, 4);
   298→   298→      ctx.globalAlpha = 1;
   299→   299→    }
   300→   300→    // ---- 5) anéis de efeito (atrás das entidades) ----
   301→   301→    this._drawRings(delta);
   302→   302→    // ---- 5.5) spotlight (separação entidade/fundo) ----
   303→   303→    var spot = ctx.createRadialGradient(300, 650, 40, 300, 650, 330);
   304→   304→    spot.addColorStop(0, 'rgba(0,0,0,0.42)');
   305→   305→    spot.addColorStop(1, 'rgba(0,0,0,0)');
   306→   306→    ctx.fillStyle = spot;
   307→   307→    ctx.fillRect(0, 330, W, 470);
   308→   308→    // ---- 6) entidades ----
   309→   309→    this._drawEntities(delta);
   310→   310→    // ---- 7) partículas de combate ----
   311→   311→    this._drawParts(delta);
   312→   312→    // ---- 8) barras de vida ----
   313→   313→    this._bar(ctx, 20, 470, 230, 30, this.heroHpShown, this.heroGhost, '#3a9e5f', '#7a3030', this.heroHpLabel, '#58e07a');
   314→   314→    this._bar(ctx, 290, 470, 230, 30, this.enemyHpShown, this.enemyGhost, '#d0455f', '#7a3030', this.enemyHpLabel, '#ff7a8a');
   315→   315→    this.heroGhost = this._ghost(this.heroGhost, this.heroHpShown, delta);
   316→   316→    this.enemyGhost = this._ghost(this.enemyGhost, this.enemyHpShown, delta);
   317→   317→    // ---- 9) dano flutuante ----
   318→   318→    this._drawFloats(delta);
   319→   319→    // ---- 10) título da região ----
   320→   320→    if (this.regionTitleT > 0) {
   321→   321→      this.regionTitleT -= delta;
   322→   322→      var a = Math.min(1, Math.min(this.regionTitleT, 2.4 - this.regionTitleT) * 2);
   323→   323→      ctx.globalAlpha = E.U.clamp(a, 0, 1) * 0.9;
   324→   324→      ctx.font = '700 26px ' + (this.fontReady ? 'Cinzel,' : '') + ' "Courier New",monospace';
   325→   325→      ctx.textAlign = 'center';
   326→   326→      ctx.fillStyle = '#000';
   327→   327→      ctx.fillText((this._regionName() || '').toUpperCase(), W / 2 + 2, 128 + 2);
   328→   328→      ctx.fillStyle = this.pal ? this.pal.accent : '#e8a33a';
   329→   329→      ctx.fillText((this._regionName() || '').toUpperCase(), W / 2, 128);
   330→   330→      ctx.globalAlpha = 1;
   331→   331→    }
