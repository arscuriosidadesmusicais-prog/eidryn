   286→   286→  _shiftHex: function(hex, S, k, br){
   287→   287→    var r = parseInt(hex.substr(1, 2), 16), g = parseInt(hex.substr(3, 2), 16), b = parseInt(hex.substr(5, 2), 16);
   288→   288→    var lum = (r * 0.3 + g * 0.59 + b * 0.11) / 255;
   289→   289→    var t = S.tint, w = S.amt * k;
   290→   290→    r = (r * br) * (1 - w * 0.7) + t[0] * w * 0.7 * (0.35 + lum * 0.65);
   291→   291→    g = (g * br) * (1 - w * 0.7) + t[1] * w * 0.7 * (0.35 + lum * 0.65);
   292→   292→    b = (b * br) * (1 - w * 0.7) + t[2] * w * 0.7 * (0.35 + lum * 0.65);
   293→   293→    return 'rgb(' + (r > 255 ? 255 : r | 0) + ',' + (g > 255 ? 255 : g | 0) + ',' + (b > 255 ? 255 : b | 0) + ')';
   294→   294→  },
   295→   295→  _tintImg: function(img, S){

   298→   298→    c.width = img.naturalWidth; c.height = img.naturalHeight;
   299→   299→    var cx = c.getContext('2d');
   300→   300→    cx.drawImage(img, 0, 0);
   301→   301→    try {
   302→   302→      var id = cx.getImageData(0, 0, c.width, c.height), px = id.data;
   303→   303→      var t = S.tint, sat = S.sat, br = S.bright, amt = S.amt;
   304→   304→      for (var i = 0; i < px.length; i += 4) {
   305→   305→        if (!px[i + 3]) continue;
   306→   306→        var r = px[i], g = px[i + 1], b = px[i + 2];
   307→   307→        var lum = r * 0.3 + g * 0.59 + b * 0.11;
   308→   308→        r = lum + (r - lum) * sat; g = lum + (g - lum) * sat; b = lum + (b - lum) * sat;
   309→   309→        r *= br; g *= br; b *= br;
   310→   310→        var w = amt * (0.3 + lum / 255 * 0.7);

   761→   761→  _drawParts: function(delta){

   788→   788→        ctx.lineTo(p.x - p.vx * 0.03, p.y - p.vy * 0.03);

   791→   791→        ctx.fillStyle = p.color;
