   256→  /* ================= FSM DO HERÓI ================= */
   257→  _heroPose: function(p, force){
   258→    var def = this.POSES[p], frames = this.heroFrames[p];
   259→    if (!def || !frames || !frames.length) return;
   260→    var curP = this.heroAnim.pose, cur = this.POSES[curP];
   261→    if (!force && cur && def.prio < cur.prio) {
   262→      var curFr = this.heroFrames[curP] || [];
   263→      var curDone = cur.loop ? false : (this.heroAnim.t * cur.fps >= curFr.length);
   264→      if (!curDone) return; // não interrompe postura mais prioritária
   265→    }
   266→    if (!force && p === curP && this.heroAnim.t < 0.05) return;
   267→    this.heroAnim.pose = p; this.heroAnim.f = 0; this.heroAnim.t = 0;
   268→  },
   269→  _heroTick: function(delta){
   270→    var a = this.heroAnim, def = this.POSES[a.pose], fr = this.heroFrames[a.pose];
   271→    if (!def || !fr || !fr.length) { a.pose = 'idle'; a.f = 0; a.t = 0; return; }
   272→    a.t += delta;
   273→    var idx = Math.floor(a.t * def.fps);
   274→    if (def.loop) { a.f = idx % fr.length; }
   275→    else if (idx >= fr.length) {
   276→      if (def.hold) { a.f = fr.length - 1; } // segura até evento (spawn/respawn)
   277→      else {
   278→        var nx = def.next && this.heroFrames[def.next] ? def.next : 'idle';
   279→        a.pose = nx; a.t = 0; a.f = 0;
   280→      }
   281→    } else a.f = idx;
   282→  },

   354→  _shiftHex: function(hex, S, k, br){
   355→    var r = parseInt(hex.substr(1, 2), 16), g = parseInt(hex.substr(3, 2), 16), b = parseInt(hex.substr(5, 2), 16);
   356→    var lum = (r * 0.3 + g * 0.59 + b * 0.11) / 255;
   357→    var t = S.tint, w = S.amt * k;
   358→    r = (r * br) * (1 - w * 0.7) + t[0] * w * 0.7 * (0.35 + lum * 0.65);
   359→    g = (g * br) * (1 - w * 0.7) + t[1] * w * 0.7 * (0.35 + lum * 0.65);
   360→    b = (b * br) * (1 - w * 0.7) + t[2] * w * 0.7 * (0.35 + lum * 0.65);
   361→    return 'rgb(' + (r > 255 ? 255 : r | 0) + ',' + (g > 255 ? 255 : g | 0) + ',' + (b > 255 ? 255 : b | 0) + ')';

   363→  _tintImg: function(img, S){
   364→    if (!img || !img.complete || !img.naturalWidth || !S) return null;
   365→    var c = document.createElement('canvas');
   366→    c.width = img.naturalWidth; c.height = img.naturalHeight;
   367→    var cx = c.getContext('2d');
   368→    cx.drawImage(img, 0, 0);
   369→    try {
   370→      var id = cx.getImageData(0, 0, c.width, c.height), px = id.data;
   371→      var t = S.tint, sat = S.sat, br = S.bright, amt = S.amt;
   372→      for (var i = 0; i < px.length; i += 4) {
   373→        if (!px[i + 3]) continue;
   374→        var r = px[i], g = px[i + 1], b = px[i + 2];
   375→        var lum = r * 0.3 + g * 0.59 + b * 0.11;
   376→        r = lum + (r - lum) * sat; g = lum + (g - lum) * sat; b = lum + (b - lum) * sat;
   377→        r *= br; g *= br; b *= br;
   378→        var w = amt * (0.3 + lum / 255 * 0.7);
   379→        r = r * (1 - w) + t[0] * w * 1.06;
   380→        g = g * (1 - w) + t[1] * w * 1.06;
   381→        b = b * (1 - w) + t[2] * w * 1.06;
   382→        px[i] = r > 255 ? 255 : r | 0; px[i + 1] = g > 255 ? 255 : g | 0; px[i + 2] = b > 255 ? 255 : b | 0;
   383→      }
   384→      cx.putImageData(id, 0, 0);
   385→    } catch (e) { return c; }
   386→    return c;

   429→  weatherNow: function(){
   430→    // escolha determinística por região + janela de 2,5min → o mundo respira sozinho
   431→    var slot = Math.floor(Date.now() / 150000);
   432→    var h = this._hashStr(this.regionId + ':wx:' + slot);
   433→    var W = this.REGION_WEATHER[this.regionId] || this.REGION_WEATHER.bosque_vidro;
   434→    var total = 0, k;
   435→    for (k in W) total += W[k];
   436→    var r = (h % 10000) / 10000 * total, acc = 0;
   437→    var wk = 'limpo';
   438→    for (k in W) { acc += W[k]; if (r < acc) { wk = k; break; } }
   439→    // coerência sazonal: no inverno, chuva vira neve (tempestade permanece — gelo)
   440→    if (wk === 'chuva' && this.seasonKey() === 'inverno') wk = 'neve';
   441→    return wk;

   469→  _weatherTick: function(delta){

   471→    if (this.weatherPref === 'auto' && this.weatherT >= 40) {
   472→      this.weatherT = 0;
   473→      var wk = this.weatherNow();
   474→      if (wk !== this.weather) this._applyWeather(false);
   475→    }

   478→    // vento com rajadas (afeta chuva, neve e partículas sazonais)

   480→    this.wind = W.wind * (0.75 + 0.25 * Math.sin(this.time * 0.6)) + Math.max(0, gust);
   481→    // relâmpagos
   482→    if (W.bolt) {
   483→      this.boltNext -= delta;
   484→      if (this.boltNext <= 0) {
   485→        this.boltNext = 2.6 + Math.random() * 5.4;
   486→        var near = Math.random() < 0.45;
   487→        if (near) {
   488→          this.boltPts = this._boltPath();
   489→          this.boltT = 0.26;
   490→          this.flashA = 0.55;
   491→          this.addShake(0.34);
   492→        } else {
   493→          this.boltPts = null;
   494→          this.flashA = 0.20;
   495→          this.addShake(0.08);
   496→        }
   497→      }
   498→    }
   499→    if (this.boltT > 0) this.boltT -= delta;

   506→  _rainTick: function(delta){
   507→    var W = this.WEATHERS[this.weather] || this.WEATHERS.limpo;
   508→    var def = W.rain || W.snow;
   509→    var target = def ? Math.round(def.n * (0.35 + 0.65 * this.weatherBlend)) : 0;
   510→    // repovoamento/esvaziamento gradual (sem pop)
   511→    if (this.rainOn < target) this.rainOn = Math.min(target, this.rainOn + 5);
   512→    else if (this.rainOn > target) this.rainOn = Math.max(target, this.rainOn - 8);
   513→    var isSnow = !!W.snow, isAsh = this._rainStyle().mode === 'ash';
   514→    var alive = 0;
   515→    for (var i = 0; i < this.rain.length; i++) {
   516→      var p = this.rain[i];
   517→      if (!p.on) {
   518→        if (alive < this.rainOn) { this._resetDrop(p, true); alive++; }
   519→        continue;
   520→      }
   521→      var wrapped = false;
   522→      if (isSnow || isAsh) {
   523→        // neve / cinzas: deriva suave com vento
   524→        p.y += p.sp * delta * (isAsh ? 0.35 : 0.45);
   525→        p.x += (Math.sin(this.time * 1.2 + p.ph) * 16 + this.wind * 46) * delta;
   526→        if (p.y > this.GROUND_Y + 6) wrapped = true;
   527→        if (p.x > this.W + 8) { p.x = -8; }
   528→      } else {
   529→        p.y += p.sp * delta;
   530→        p.x += this.wind * 175 * delta;
   531→        if (p.y > this.GROUND_Y - 4 + p.gy) {
   532→          if (p.front && Math.random() < 0.34) this._splash(p.x, this.GROUND_Y + 2 + p.gy);
   533→          wrapped = true;
   534→        }
   535→        if (p.x > this.W + 12) p.x = -10;
   536→      }
   537→      if (wrapped) {
   538→        alive++;
   539→        if (alive > this.rainOn) { p.on = false; continue; }
   540→        this._resetDrop(p, true);
   541→      } else alive++;
   542→    }
   543→    // splashes do solo
   544→    for (var s = 0; s < this.splashes.length; s++) {
   545→      var sp = this.splashes[s];
   546→      if (!sp.on) continue;
   547→      sp.t += delta;
   548→      if (sp.t >= sp.life) { sp.on = false; continue; }
   549→    }
   550→  },
   551→  _resetDrop: function(p, keepOn){

   553→    var def = W.rain || W.snow || {};

   555→    p.on = keepOn !== false;
   556→    p.x = Math.random() * (this.W + 90) - 45;
   557→    p.y = -12 - Math.random() * 60;
   558→    p.sp = sp[0] + Math.random() * (sp[1] - sp[0]);
   559→    p.ph = Math.random() * 6.28;
   560→    p.front = Math.random() < 0.32;
   561→    p.gy = Math.random() * 26;
