     9→/* ---------- PET (6 pets + 4 companheiros, estrelas) ---------- */
    10→E.Pet = {
    11→  owned: {}, fragments: {}, active_pet: '', active_companion: '',
    12→  all_defs: function(){ return E.DM.cfg_pets.pets.concat(E.DM.cfg_pets.companions); },
    13→  def: function(id){
    14→    var all=this.all_defs();
    15→    for(var i=0;i<all.length;i++) if(all[i].id===id) return all[i];
    16→    return {};
    17→  },
    18→  is_owned: function(id){ return id in this.owned; },
    19→  stars: function(id){ return this.owned[id]||0; },
    20→  acquire: function(id){
    21→    if(id in this.owned){
    22→      var qty=E.DM.cfg_gacha.pet_duplicate.frag_qty;
    23→      this.fragments[id]=(this.fragments[id]||0)+qty;
    24→      E.BUS.toast_msg(this.def(id).name+' → +'+qty+' fragmentos', '#3a9e8f');
    25→      E.Save.mark_dirty();
    26→      return 'duplicate';
    27→    }
    28→    this.owned[id]=1;
    29→    if(!(id in this.fragments)) this.fragments[id]=0;
    30→    var d=this.def(id);
    31→    if(d.type==='pet' && this.active_pet==='') this.active_pet=id;
    32→    if(d.type==='companheiro' && this.active_companion==='') this.active_companion=id;
    33→    E.Char.recalc();
    34→    E.BUS.emit('pet_changed');
    35→    E.Save.mark_dirty();
    36→    return 'new';
    37→  },
    38→  set_active: function(id){
    39→    var d=this.def(id);
    40→    if(!d.id || !this.is_owned(id)) return;
    41→    if(d.type==='pet') this.active_pet=id; else this.active_companion=id;
    42→    E.Char.recalc();
    43→    E.BUS.emit('pet_changed');
    44→    E.Save.mark_dirty();
    45→  },
    46→  pet_bonus: function(id){
    47→    var b={}, d=this.def(id);
    48→    if(!d.id || !this.is_owned(id)) return b;
    49→    var st=this.stars(id);
    50→    for(var k in d.bonus) b[k]=d.bonus[k]+d.per_star[k]*(st-1);
    51→    return b;
    52→  },
    53→  all_bonuses: function(){
    54→    var b={};
    55→    if(this.active_pet && this.is_owned(this.active_pet)) this._merge(this.pet_bonus(this.active_pet), b);
    56→    if(this.active_companion && this.is_owned(this.active_companion)) this._merge(this.pet_bonus(this.active_companion), b);
    57→    return b;
    58→  },
    59→  _merge: function(src, dst){ for(var k in src) dst[k]=(dst[k]||0)+src[k]; },
    60→  bonus_value: function(key){ return this.all_bonuses()[key]||0; },
    61→  evolve_cost_frags: function(id){
    62→    var next=this.stars(id)+1, c=E.DM.cfg_pets.star_costs;
    63→    if(next>E.DM.cfg_pets.max_stars || c[String(next)]==null) return 0;
    64→    return c[String(next)];
    65→  },
    66→  evolve_cost_gold: function(id){
    67→    var next=this.stars(id)+1, c=E.DM.cfg_pets.star_gold;
    68→    if(next>E.DM.cfg_pets.max_stars || c[String(next)]==null) return 0;
    69→    return c[String(next)];
    70→  },
    71→  evolve: function(id){
    72→    if(!this.is_owned(id)) return false;
    73→    var need_f=this.evolve_cost_frags(id);
    74→    if(need_f<=0){ E.BUS.toast_msg(E.DM.tr('already_max'), '#9aa0a6'); return false; }
    75→    var need_g=this.evolve_cost_gold(id);
    76→    if((this.fragments[id]||0)<need_f || !E.Eco.can_spend('ouro', need_g)){
    77→      E.BUS.toast_msg(E.DM.tr('not_enough'), '#d0455f');
    78→      return false;
    79→    }
    80→    this.fragments[id]-=need_f;
    81→    E.Eco.spend('ouro', need_g);
    82→    this.owned[id]=this.stars(id)+1;
    83→    E.Char.recalc();
    84→    E.BUS.emit('pet_star_up', id, this.owned[id]);
    85→    E.Save.mark_dirty();
    86→    return true;
    87→  },
    88→  add_fragments: function(id, qty){ this.fragments[id]=(this.fragments[id]||0)+qty; E.Save.mark_dirty(); },
    89→  owned_count: function(){ return Object.keys(this.owned).length; },
    90→  max_stars_any: function(){ var m=0; for(var id in this.owned) m=Math.max(m,this.owned[id]); return m; },
    91→  save_state: function(){ return {owned:this.owned, fragments:this.fragments, active_pet:this.active_pet, active_companion:this.active_companion}; },
    92→  load_state: function(d){
    93→    if(!d) return;
    94→    this.owned=d.owned||{}; this.fragments=d.fragments||{};
    95→    this.active_pet=d.active_pet||''; this.active_companion=d.active_companion||'';
    96→  }
    97→};

    99→/* ---------- GACHA (Portal do Crepúsculo, pity transparente 10/50/100) ---------- */
   100→E.Gacha = {
   101→  pulls_total: 0, pulls_since_rare: 0, pulls_since_epic: 0, pulls_since_legend: 0,
   102→  pity_state: function(){ return {total:this.pulls_total, since_rare:this.pulls_since_rare,
   103→    since_epic:this.pulls_since_epic, since_legend:this.pulls_since_legend}; },
   104→  pull: function(n){
   105→    var cost_per=E.DM.cfg_gacha.pull_cost, cost10=E.DM.cfg_gacha.pull10_cost;
   106→    var total_cost=(n===10)?cost10:cost_per*n;
   107→    if(!E.Eco.spend('essencia', total_cost)) return [];
   108→    var results=[], bonus=(n===10 && E.DM.cfg_gacha.pull10_bonus)?1:0;
   109→    for(var i=0;i<n+bonus;i++) results.push(this._single_pull());
   110→    E.Ret.track('gacha', n);
   111→    E.Save.mark_dirty();
   112→    return results;
   113→  },
   114→  _single_pull: function(){
   115→    this.pulls_total++; this.pulls_since_rare++; this.pulls_since_epic++; this.pulls_since_legend++;
   116→    /* Pity (D10 + HTML-1): 10 → Rara+, 50 → Épica+, 100 → Lendária+ */
   117→    var min_rarity='';
   118→    if(this.pulls_since_legend>=100) min_rarity='lendaria';
   119→    else if(this.pulls_since_epic>=50) min_rarity='epica';
   120→    else if(this.pulls_since_rare>=10) min_rarity='rara'; // [HTML-1]
   121→    var rarity=this._roll_rarity(min_rarity);
   122→    if(['rara','epica','lendaria','mitica','divina'].indexOf(rarity)>=0) this.pulls_since_rare=0;
   123→    if(['epica','lendaria','mitica','divina'].indexOf(rarity)>=0) this.pulls_since_epic=0;
   124→    if(['lendaria','mitica','divina'].indexOf(rarity)>=0) this.pulls_since_legend=0;
   125→    var type_roll=Math.random()*100, types=E.DM.cfg_gacha.types;
   126→    var pet_w=types.pet, comp_w=types.companheiro;
   127→    if(type_roll<pet_w) return this._pull_pet(rarity);
   128→    else if(type_roll<pet_w+comp_w) return this._pull_companion(rarity);
   129→    else return this._pull_item(rarity);
   130→  },
   131→  _roll_rarity: function(min_rarity){
   132→    var rates=E.DM.cfg_gacha.rates, order=['rara','epica','lendaria','mitica','divina'];
   133→    var min_idx=order.indexOf(min_rarity); if(min_idx<0) min_idx=0;
   134→    var weights=[], total=0;
   135→    for(var i=0;i<order.length;i++){
   136→      var w=Number(rates[order[i]]);
   137→      if(i<min_idx) w=0;
   138→      weights.push(w); total+=w;
   139→    }
   140→    var roll=Math.random()*total, acc=0;
   141→    for(var j=0;j<weights.length;j++){ acc+=weights[j]; if(roll<=acc) return order[j]; }
   142→    return order[min_idx];
   143→  },
   144→  _pull_pet: function(rarity){
   145→    var pool=[], pets=E.DM.cfg_pets.pets;
   146→    for(var i=0;i<pets.length;i++) if(E.Inv.rarity_order(pets[i].rarity)>=E.Inv.rarity_order(rarity)) pool.push(pets[i]);
   147→    if(pool.length===0) pool=pets.slice();
   148→    var p=pool[Math.floor(Math.random()*pool.length)];
   149→    var res=E.Pet.acquire(p.id);
   150→    return {kind:'pet', id:p.id, name:p.name, rarity:p.rarity, detail:res};
   151→  },
   152→  _pull_companion: function(rarity){
   153→    var pool=[], comps=E.DM.cfg_pets.companions;
   154→    for(var i=0;i<comps.length;i++) if(E.Inv.rarity_order(comps[i].rarity)>=E.Inv.rarity_order(rarity)) pool.push(comps[i]);
   155→    if(pool.length===0) pool=comps.slice();
   156→    var c=pool[Math.floor(Math.random()*pool.length)];
   157→    var res=E.Pet.acquire(c.id);
   158→    return {kind:'companheiro', id:c.id, name:c.name, rarity:c.rarity, detail:res};
   159→  },
   160→  _pull_item: function(rarity){
   161→    var item_rates=E.DM.cfg_gacha.item_pull_rates;
   162→    var order=['rara','epica','lendaria','mitica','divina'];
   163→    var min_idx=order.indexOf(rarity); if(min_idx<0) min_idx=0;
   164→    /* [HTML-1] pity aplica-se também a pulls de item: pesos abaixo do piso são zerados */
   165→    var weights=[], total=0;
   166→    for(var i=0;i<order.length;i++){
   167→      var w= i<min_idx? 0 : Number(item_rates[order[i]]);
   168→      weights.push(w); total+=w;
   169→    }
   170→    var roll=Math.random()*total, acc=0, chosen=order[min_idx];
   171→    for(var j=0;j<weights.length;j++){ acc+=weights[j]; if(roll<=acc){ chosen=order[j]; break; } }
   172→    var it=E.Inv.generate_item(E.Prog.farm_stage()+5, chosen);
   173→    E.Inv.add_item(it);
   174→    return {kind:'item', id:it.id, name:it.base, rarity:chosen, detail:it};
   175→  },
   176→  save_state: function(){ return {pulls_total:this.pulls_total, pulls_since_rare:this.pulls_since_rare,
   177→    pulls_since_epic:this.pulls_since_epic, pulls_since_legend:this.pulls_since_legend}; },
   178→  load_state: function(d){
   179→    if(!d) return;
   180→    this.pulls_total=d.pulls_total||0; this.pulls_since_rare=d.pulls_since_rare||0;
   181→    this.pulls_since_epic=d.pulls_since_epic||0; this.pulls_since_legend=d.pulls_since_legend||0;
   182→  }
   183→};

   185→/* ---------- PROGRESSION (500 fases, 6 regiões + Abismo, farm fallback) ---------- */
   186→E.Prog = {
   187→  current_stage: 1, max_stage: 1, auto_advance: true, bosses_killed_total: 0, _region_id: '',
   188→  init: function(){
   189→    var self=this;
   190→    E.BUS.on('combat_ended', function(result, stage){ self._on_combat_ended(result, stage); });
   191→  },
   192→  current_region: function(){ return E.DM.region_for_stage(this.current_stage); },
   193→  farm_stage: function(){
   194→    if(this.max_stage%10===0) return Math.max(1, this.max_stage-1);
   195→    return this.max_stage;
   196→  },
   197→  set_auto_advance: function(on){
   198→    this.auto_advance=on;
   199→    E.BUS.emit('auto_advance_toggled', on);
   200→    E.Save.mark_dirty();
   201→  },
   202→  jump_to_stage: function(stage){
   203→    this.current_stage=E.U.clamp(stage, 1, Math.max(this.max_stage, this.current_stage));
   204→    this._notify_stage();
   205→  },
   206→  _on_combat_ended: function(result, stage){
   207→    if(E.Combat.mode!=='campaign') return;
   208→    if(result==='win'){
   209→      if(stage>this.max_stage) this.max_stage=stage;
   210→      if(E.DM.is_boss_stage(stage)) this.bosses_killed_total++;
   211→      if(this.auto_advance) this.current_stage=stage+1;
   212→      this._notify_stage();
   213→    } else {
   214→      this.current_stage=this.farm_stage();
   215→      E.BUS.emit('farming_fallback', this.current_stage);
   216→      this._notify_stage();
   217→    }
   218→  },
   219→  _notify_stage: function(){
   220→    var is_boss=E.DM.is_boss_stage(this.current_stage), is_mini=E.DM.is_miniboss_stage(this.current_stage);
   221→    E.BUS.emit('stage_changed', this.current_stage, is_boss, is_mini);
   222→    var reg=this.current_region();
   223→    if(reg.id!==this._region_id){ this._region_id=reg.id; E.BUS.emit('region_changed', reg.id); }
   224→    E.Save.mark_dirty();
   225→  },
   226→  /* Rendimento/segundo na fase de farm (usado por Offline e Masmorras) */
   227→  gold_per_second: function(stage){
   228→    var e=E.DM.enemy_stats_for_stage(stage);
   229→    var interval=E.Char.stats().atk_interval||1;
   230→    var kill_time=interval*4+0.6;
   231→    return e.gold/Math.max(kill_time,0.5)*E.Eco.gold_mult();
   232→  },
   233→  xp_per_second: function(stage){
   234→    var e=E.DM.enemy_stats_for_stage(stage);
   235→    var interval=E.Char.stats().atk_interval||1;
   236→    var kill_time=interval*4+0.6;
   237→    return e.xp/Math.max(kill_time,0.5);
   238→  },
   239→  save_state: function(){ return {current_stage:this.current_stage, max_stage:this.max_stage,
   240→    auto_advance:this.auto_advance, bosses_killed_total:this.bosses_killed_total}; },
   241→  load_state: function(d){
   242→    if(!d) return;
   243→    this.current_stage=d.current_stage||1; this.max_stage=d.max_stage||1;
   244→    this.auto_advance=d.auto_advance!==false; this.bosses_killed_total=d.bosses_killed_total||0;
   245→    this._region_id=this.current_region().id;
   246→  }
   247→};

   249→/* ---------- ASCENSION (reinício do ciclo com árvore permanente) ---------- */
   250→E.Asc = {
   251→  ascensions: 0, node_levels: {},
   252→  unlock_stage: function(){ return E.DM.cfg_ascension.unlock_stage; },
   253→  is_unlocked: function(){ return E.Prog.max_stage>=this.unlock_stage(); },
   254→  pending_fragments: function(){
   255→    if(!this.is_unlocked()) return 0;
   256→    var fs=Math.max(E.Prog.farm_stage(),1);
   257→    return Math.floor(Math.pow(fs/10, 1.35))+E.Prog.bosses_killed_total;
   258→  },
   259→  can_ascend: function(){ return this.is_unlocked() && this.pending_fragments()>0; },
   260→  ascend: function(){
   261→    if(!this.can_ascend()) return false;
   262→    var gained=this.pending_fragments();
   263→    E.Eco.add('fragmentos_alma', gained);
   264→    E.Prog.current_stage=1; E.Prog.max_stage=1; E.Prog.bosses_killed_total=0;
   265→    E.Char.level=1; E.Char.xp=0;
   266→    for(var a in E.Char.attributes) E.Char.attributes[a]=0;
   267→    E.Eco.currencies.ouro=E.DM.cfg_currencies.starting.ouro;
   268→    this.ascensions++;
   269→    E.Char.recalc();
   270→    E.Combat.stop();
   271→    E.Combat.start_campaign(1);
   272→    E.BUS.emit('ascension_performed', gained);
   273→    E.BUS.toast_msg('☾ CICLO ENCERRADO — +'+Math.floor(gained)+' Fragmentos de Alma ☽', '#f5e6c8');
   274→    E.Save.flush();
   275→    return true;
   276→  },
   277→  nodes: function(){ return E.DM.cfg_ascension.nodes; },
   278→  node_level: function(id){ return this.node_levels[id]||0; },
   279→  node_cost: function(n){ return n.cost*(this.node_level(n.id)+1); },
   280→  node_requirements_met: function(n){
   281→    var req=n.requires||[];
   282→    for(var i=0;i<req.length;i++) if(this.node_level(req[i])<=0) return false;
   283→    return true;
   284→  },
   285→  buy_node: function(id){
   286→    var n=null, list=this.nodes();
   287→    for(var i=0;i<list.length;i++) if(list[i].id===id){ n=list[i]; break; }
   288→    if(!n) return false;
   289→    var lv=this.node_level(id);
   290→    if(lv>=n.max){ E.BUS.toast_msg(E.DM.tr('already_max'), '#9aa0a6'); return false; }
   291→    if(!this.node_requirements_met(n)){ E.BUS.toast_msg('Requisitos não atendidos', '#d0455f'); return false; }
   292→    if(!E.Eco.spend('fragmentos_alma', this.node_cost(n))) return false;
   293→    this.node_levels[id]=lv+1;
   294→    E.Char.recalc();
   295→    E.BUS.emit('node_bought', id);
   296→    E.Save.mark_dirty();
   297→    return true;
   298→  },
   299→  tree_bonuses: function(){
   300→    var b={}, list=this.nodes();
   301→    for(var i=0;i<list.length;i++){
   302→      var lv=this.node_level(list[i].id);
   303→      if(lv<=0) continue;
   304→      for(var k in list[i].bonus) b[k]=(b[k]||0)+list[i].bonus[k]*lv;
   305→    }
   306→    return b;
   307→  },
   308→  has_revive: function(){ return this.node_level('n10')>0; },
   309→  offline_cap_bonus_hours: function(){ return this.tree_bonuses().offline_hours||0; },
   310→  save_state: function(){ return {ascensions:this.ascensions, node_levels:this.node_levels}; },
   311→  load_state: function(d){
   312→    if(!d) return;
   313→    this.ascensions=d.ascensions||0; this.node_levels=d.node_levels||{};
   314→  }
   315→};

   317→/* ---------- MODES (Masmorras, Torre, World Boss, Arena) ---------- */
   318→E.Modes = {
   319→  tower_floor: 1, tower_record: 0, wb_damage: 0, wb_week: '', wb_last_rank: 0,
   320→  arena_points: 0, arena_wins_total: 0,
   321→  dungeon_daily: {}, cooldowns: {}, arena_attempts_today: {},
   322→  _active_mode: '', _active_t: 0, _active_dur: 0, _wb_active: false,
   323→  init: function(){
   324→    var self=this;
   325→    E.BUS.on('enemy_damaged', function(amount){ if(self._wb_active) self.wb_damage+=amount; });
   326→  },
   327→  can_enter: function(mode_id){
   328→    var d=E.DM.cfg_dungeons[mode_id];
   329→    if(!d) return {ok:false, reason:'modo desconhecido'};
   330→    if(mode_id==='arena'){
   331→      var used=(this.arena_attempts_today[E.TimeM.day_key()]||0);
   332→      if(used>=d.attempts_free) return {ok:false, reason:'sem tentativas hoje', extra_gems:d.extra_attempt_gems};
   333→      return {ok:true};
   334→    }
   335→    var free_key='free_'+mode_id;
   336→    var used_free=((this.dungeon_daily[E.TimeM.day_key()]||{})[free_key]||0);
   337→    var free_daily=d.free_daily||0;
   338→    if((mode_id==='masmorra_ouro'||mode_id==='masmorra_xp') && used_free<free_daily) return {ok:true, free:true};
   339→    if((d.key_cost||0)>0 && E.Eco.get_cur('chaves')<d.key_cost) return {ok:false, reason:'sem chaves'};
   340→    var cd=this.cooldowns[mode_id]||0;
   341→    if(E.TimeM.now()<cd) return {ok:false, reason:'em recarga', ready_at:cd};
   342→    return {ok:true, free:false};
   343→  },
   344→  enter: function(mode_id){
   345→    var check=this.can_enter(mode_id);
   346→    if(!check.ok){ E.BUS.toast_msg(check.reason||'', '#d0455f'); return false; }
   347→    var d=E.DM.cfg_dungeons[mode_id];
   348→    if(mode_id==='arena'){
   349→      var k=E.TimeM.day_key();
   350→      this.arena_attempts_today[k]=(this.arena_attempts_today[k]||0)+1;
   351→      return this._resolve_arena();
   352→    }
   353→    if(mode_id==='masmorra_ouro'||mode_id==='masmorra_xp'){
   354→      var free=!!check.free;
   355→      if(!free && (d.key_cost||0)>0){ if(!E.Eco.spend('chaves', d.key_cost)) return false; }
   356→      else this._track_daily('free_'+mode_id);
   357→    } else if(mode_id==='masmorra_equip'){
   358→      if(!E.Eco.spend('chaves', d.key_cost!=null?d.key_cost:1)) return false;
   359→    }
   360→    if(mode_id==='masmorra_ouro'||mode_id==='masmorra_xp'||mode_id==='masmorra_equip'){
   361→      this._active_mode=mode_id; this._active_t=0; this._active_dur=d.duration_s;
   362→      E.BUS.emit('dungeon_started', mode_id, d);
   363→      return true;
   364→    }
   365→    if(mode_id==='torre_infinita'){ this.tower_floor=1; this._start_tower_floor(); return true; }
   366→    if(mode_id==='world_boss'){ this._start_world_boss(); return true; }
   367→    return false;
   368→  },
   369→  _track_daily: function(key){
   370→    var k=E.TimeM.day_key();
   371→    if(!this.dungeon_daily[k]) this.dungeon_daily[k]={};
   372→    this.dungeon_daily[k][key]=(this.dungeon_daily[k][key]||0)+1;

   374→  process: function(delta){
