     7→/* ---------- COMBAT (loop automático, skills, modificadores, recompensas) ---------- */
     8→E.Combat = {
     9→  ENEMY_INTERVAL: 2.0,
    10→  RESPAWN_DELAY: 0.7,
    11→  active: false,
    12→  mode: 'campaign',
    13→  stage: 1,
    14→  enemy: {}, enemy_hp: 0, enemy_shield: 0, enemy_berserk: false,
    15→  hero_hp: 0, hero_shield: 0, revive_used: false,
    16→  _hero_t: 0, _enemy_t: 0, _spawn_t: 0, _boss_t: 0,
    17→  _buff_atk_pct: 0, _buff_time: 0,
    18→  _dot_dmg: 0, _dot_time: 0, _dot_per_s: 0, _dot_lifesteal: false,
    19→  _enemy_atk_shred: 0, _enemy_def_shred: 0, _stun_t: 0, _reflect_pct: 0,
    20→  _mode_callback: '', _restart_pending: false,
    21→
    22→  start_campaign: function(stage){
    23→    this.mode='campaign'; this.stage=stage; this._begin();
    24→  },
    25→  start_custom: function(ctx){
    26→    this.mode=ctx.mode||'tower';
    27→    this.stage=ctx.stage||1;
    28→    this.enemy=ctx.enemy||{};
    29→    this._mode_callback=ctx.callback||'';
    30→    this._begin_with_enemy();
    31→  },
    32→  stop: function(){ this.active=false; },
    33→  _begin: function(){
    34→    this.enemy=E.DM.enemy_for_stage(this.stage);
    35→    this._begin_with_enemy();
    36→  },
    37→  _begin_with_enemy: function(){
    38→    var stats=E.Char.stats();
    39→    this.hero_hp=stats.hp; this.hero_shield=0; this.revive_used=false;
    40→    this._reset_effects();
    41→    this.active=true;
    42→    this._spawn_t=0; this._hero_t=0; this._enemy_t=0; this._restart_pending=false;
    43→    this.enemy_hp=this._enemy_max_hp();
    44→    if(this.enemy.boss) this._boss_t=E.DM.cfg_enemies.boss_timer_s;
    45→    else this._boss_t=0;
    46→    this._apply_spawn_modifiers();
    47→    E.BUS.emit('combat_started', this.stage, !!this.enemy.boss);
    48→    E.BUS.emit('enemy_spawned', this.enemy);
    49→    E.BUS.emit('enemy_hp_changed', this.enemy_hp+this.enemy_shield, this._enemy_max_hp());
    50→    E.BUS.emit('hero_hp_changed', this.hero_hp, stats.hp);
    51→  },
    52→  _reset_effects: function(){
    53→    this._buff_atk_pct=0; this._buff_time=0;
    54→    this._dot_dmg=0; this._dot_time=0; this._dot_per_s=0; this._dot_lifesteal=false;
    55→    this._enemy_atk_shred=0; this._enemy_def_shred=0;
    56→    this._stun_t=0; this._reflect_pct=0;
    57→  },
    58→  _apply_spawn_modifiers: function(){
    59→    this.enemy_shield=0; this.enemy_berserk=false;
    60→    var mods=this.enemy.modifiers||[];
    61→    if(mods.indexOf('escudo')>=0) this.enemy_shield=this._enemy_max_hp()*0.25;
    62→    if(mods.indexOf('reforçado')>=0) this.enemy.def=this.enemy.def*1.2;
    63→  },
    64→  _enemy_max_hp: function(){ return this.enemy.hp||1; },
    65→  _enemy_atk: function(){
    66→    var a=this.enemy.atk||1;
    67→    if(this.enemy_berserk) a*=1.5;
    68→    if(this._enemy_atk_shred>0) a*=1-this._enemy_atk_shred/100;
    69→    return a;
    70→  },
    71→  /* ---------- LOOP ---------- */
    72→  process: function(delta){
    73→    if(!this.active) return;
    74→    var stats=E.Char.stats();
    75→    if(this._spawn_t>0){
    76→      this._spawn_t-=delta;
    77→      if(this._spawn_t<=0){
    78→        if(this._restart_pending){ this.stage=E.Prog.current_stage; this._begin(); } // [HTML-5]
    79→        else this._next_after_win();
    80→      }
    81→      return;
    82→    }
    83→    if(!this._alive()) return;
    84→    this._auto_cast();
    85→    E.Skill.tick_cooldowns(delta);
    86→    if(stats.regen>0) this.hero_hp=Math.min(this.hero_hp+stats.regen*delta, stats.hp);
    87→    if(this._buff_time>0){
    88→      this._buff_time-=delta;
    89→      if(this._buff_time<=0) this._buff_atk_pct=0;
    90→    }
    91→    if(this._dot_time>0){
    92→      this._dot_time-=delta;
    93→      var tick=this._dot_per_s*delta;
    94→      this._apply_damage_to_enemy(tick, false, '#9b59d0', true);
    95→      if(this._dot_lifesteal) this._heal_hero(tick*0.5);
    96→      if(this._dot_time<=0) this._enemy_atk_shred=0;
    97→    }
    98→    if(this._stun_t>0) this._stun_t-=delta;
    99→    this._hero_t+=delta;
   100→    var interval=stats.atk_interval;
   101→    if(this._hero_t>=interval){
   102→      this._hero_t-=interval;
   103→      this._hero_attack();
   104→    }
   105→    if(this._stun_t<=0){
   106→      this._enemy_t+=delta;
   107→      if(this._enemy_t>=this.ENEMY_INTERVAL){
   108→        this._enemy_t-=this.ENEMY_INTERVAL;
   109→        this._enemy_attack(stats);
   110→      }
   111→    }
   112→    if(this._boss_t>0){
   113→      this._boss_t-=delta;
   114→      E.BUS.emit('boss_timer', this._boss_t);
   115→      if(this._boss_t<=0 && this._alive()) this._fail('timeout');
   116→    }
   117→    if(this._alive() && (this.enemy.modifiers||[]).indexOf('curandeiro')>=0){
   118→      this.enemy_hp=Math.min(this.enemy_hp+this._enemy_max_hp()*0.02*delta, this._enemy_max_hp());
   119→      E.BUS.emit('enemy_hp_changed', this.enemy_hp+this.enemy_shield, this._enemy_max_hp());
   120→    }
   121→  },
   122→  enemy_interval: function(){ return this.ENEMY_INTERVAL; },
   123→  _alive: function(){ return this.active && this.enemy_hp>0 && this.hero_hp>0; },
   124→  /* ---------- ATAQUES ---------- */
   125→  _hero_attack: function(){ this._apply_damage_to_enemy(0, false, '#e8e0d0', false); },
   126→  _enemy_attack: function(stats){
   127→    var mods=this.enemy.modifiers||[];
   128→    var dodge=stats.dodge||0;
   129→    if(Math.random()*100<=dodge){
   130→      E.BUS.emit('floating_damage', 0, false, 'hero', '#9aa0a6');
   131→      return;
   132→    }
   133→    var raw=this._enemy_atk();
   134→    var dmg=E.DM.damage_taken(raw, stats.def);
   135→    this.hero_hp-=dmg;
   136→    E.BUS.emit('hero_damaged', dmg);
   137→    E.BUS.emit('floating_damage', -dmg, false, 'hero', '#d0455f');
   138→    E.BUS.emit('screenshake', this.enemy.boss?0.25:0.1);
   139→    if(this.hero_shield>0){
   140→      var absorbed=Math.min(this.hero_shield, dmg);
   141→      this.hero_shield-=absorbed;
   142→      if(this._reflect_pct>0){
   143→        var ref=dmg*this._reflect_pct/100;
   144→        this._apply_damage_to_enemy(ref, false, '#3a7bd5', true);
   145→      }
   146→    }
   147→    E.BUS.emit('hero_hp_changed', Math.max(this.hero_hp,0), stats.hp);
   148→    if(mods.indexOf('berserk')>=0 && !this.enemy_berserk && this.enemy_hp<this._enemy_max_hp()*0.3){
   149→      this.enemy_berserk=true;
   150→      E.BUS.toast_msg((this.enemy.name||'?')+' ficou Enfurecido!', '#d0455f');
   151→    }
   152→    if(this.hero_hp<=0) this._hero_down();
   153→  },
   154→  _hero_down: function(){
   155→    if(!this.revive_used && E.Asc.has_revive()){
   156→      this.revive_used=true;
   157→      this.hero_hp=E.Char.stats().hp*0.5;
   158→      E.BUS.emit('hero_hp_changed', this.hero_hp, E.Char.stats().hp);
   159→      E.BUS.toast_msg('Sombra Guardiã — renasceu!', '#3a9e8f');
   160→      return;
   161→    }
   162→    this._fail('death');
   163→  },
   164→  _apply_damage_to_enemy: function(skill_mult, _is_skill, color_hex, no_effects){
   165→    if(!this._alive()) return;
   166→    var stats=E.Char.stats();
   167→    var mods=this.enemy.modifiers||[];
   168→    if(!no_effects && mods.indexOf('elusivo')>=0 && Math.random()*100<=10){
   169→      E.BUS.emit('floating_damage', 0, false, 'enemy', '#9aa0a6');
   170→      return;
   171→    }
   172→    var crit=E.DM.roll_crit({randf_range:function(a,b){return a+Math.random()*(b-a);}}, stats.crit_rate);
   173→    var buffs=1+this._buff_atk_pct/100;
   174→    var skill_m=skill_mult<=0?1:skill_mult;
   175→    var dmg=E.DM.damage_final(stats.atk, 1, skill_m, buffs);
   176→    if(this.enemy.boss||this.enemy.miniboss) dmg*=1+(stats.boss_damage||0)/100;
   177→    if(crit) dmg*=stats.crit_damage/100;
   178→    var edef=(this.enemy.def||0)*(1-this._enemy_def_shred/100);
   179→    dmg=E.DM.damage_taken(dmg, edef);
   180→    if(this.enemy_shield>0){
   181→      var absorbed=Math.min(this.enemy_shield, dmg);
   182→      this.enemy_shield-=absorbed;
   183→      dmg-=absorbed;
   184→    }
   185→    this.enemy_hp-=dmg;
   186→    E.BUS.emit('floating_damage', dmg, crit, 'enemy', crit?'#e8833a':color_hex);
   187→    if(crit) E.BUS.emit('screenshake', 0.3);
   188→    E.BUS.emit('enemy_damaged', dmg, crit);
   189→    E.BUS.emit('enemy_hp_changed', Math.max(this.enemy_hp,0)+this.enemy_shield, this._enemy_max_hp());
   190→    var heal=E.DM.lifesteal_heal(dmg, stats.lifesteal);
   191→    if(heal>0) this._heal_hero(heal);
   192→    if(this.enemy_hp<=0) this._kill_enemy();
   193→  },
   194→  _heal_hero: function(amount){
   195→    var max_hp=E.Char.stats().hp;
   196→    var before=this.hero_hp;
   197→    this.hero_hp=Math.min(this.hero_hp+amount, max_hp);
   198→    if(this.hero_hp-before>0.5) E.BUS.emit('hero_hp_changed', this.hero_hp, max_hp);
   199→  },
   200→  /* ---------- SKILLS ---------- */
   201→  cast_skill: function(id){
   202→    var stats=E.Char.stats();
   203→    var s=E.Skill.skill_def(id);
   204→    if(!s.id) return false;
   205→    if(!E.Skill.cast(id)) return false;
   206→    var mult=E.Skill.active_mult(id)/100;
   207→    var evolved=!!E.Skill.evolved[id];
   208→    if(id==='lamina_eclipse'){
   209→      this._apply_damage_to_enemy(mult, true, '#e8833a', false);
   210→      if(evolved) this._stun_t=s.evolve.extra_stun;
   211→    } else if(id==='guarda_sombras'){
   212→      this.hero_shield=stats.hp*mult;
   213→      this._reflect_pct=evolved?s.evolve.reflect:0;
   214→      E.BUS.toast_msg(s.name+'!', '#3a7bd5');
   215→    } else if(id==='sedenta'){
   216→      this._dot_per_s=stats.atk*mult/s.base.dur;
   217→      this._dot_time=s.base.dur;
   218→      this._dot_lifesteal=true;
   219→      if(evolved) this._enemy_atk_shred=s.evolve.atk_shred;
   220→      E.BUS.toast_msg(s.name+'!', '#9b59d0');
   221→    } else if(id==='rumo_vazio'){
   222→      this._enemy_def_shred=evolved?s.evolve.def_shred:0;
   223→      this._apply_damage_to_enemy(mult, true, '#3a9e8f', false);
   224→    } else if(id==='cataclismo'){
   225→      this._apply_damage_to_enemy(mult, true, '#f5e6c8', false);
   226→      this._buff_atk_pct=E.Skill.scaled_value(id,'buff_pct');
   227→      this._buff_time=s.base.buff_dur;
   228→      E.BUS.emit('screenshake', 0.8);
   229→      E.BUS.toast_msg('☾ '+s.name+' ☽', '#f5e6c8');
   230→    }
   231→    return true;
   232→  },
   233→  _auto_cast: function(){
   234→    var ready=E.Skill.ready_skills(true);
   235→    for(var i=0;i<ready.length;i++) this.cast_skill(ready[i].id);
   236→  },
   237→  /* ---------- MORTE / RECOMPENSAS ---------- */
   238→  _kill_enemy: function(){
   239→    this.enemy_hp=0;
   240→    var stats=E.Char.stats();
   241→    if((this.enemy.modifiers||[]).indexOf('explosivo')>=0){
   242→      var boom=(this.enemy.atk||0)*0.15;
   243→      this.hero_hp-=E.DM.damage_taken(boom, stats.def);
   244→      E.BUS.emit('floating_damage', -boom, false, 'hero', '#d0455f');
   245→      E.BUS.emit('hero_hp_changed', Math.max(this.hero_hp,0), stats.hp);
   246→      if(this.hero_hp<=0){ this._hero_down(); return; }
   247→    }
   248→    var gold=(this.enemy.gold||0)*(1+(stats.gold_find||0)/100)*E.Eco.gold_mult();
   249→    var xp=(this.enemy.xp||0)*(1+(stats.xp_gain||0)/100);
   250→    E.Eco.add('ouro', gold);
   251→    E.Char.gain_xp(xp);
   252→    E.Inv.try_drop(this.stage, !!this.enemy.boss, stats);
   253→    E.Ret.track('kills', 1);
   254→    E.Ret.bp_add_xp(E.DM.cfg_battlepass.xp_sources.kills); // [HTML-3]
   255→    if(this.enemy.boss||this.enemy.miniboss){
   256→      E.Ret.track('boss_kills', 1);
   257→      E.Ret.bp_add_xp(E.DM.cfg_battlepass.xp_sources.boss_kills); // [HTML-3]
   258→    }
   259→    E.BUS.emit('enemy_killed', this.enemy);
   260→    E.BUS.emit('combat_ended', 'win', this.stage);
   261→    this._spawn_t=this.RESPAWN_DELAY;
   262→    if(this._mode_callback!=='') E.Modes.on_mode_combat_end(this._mode_callback, 'win', this.stage);
   263→  },
   264→  _fail: function(reason){
   265→    E.BUS.emit('combat_ended', 'fail', this.stage);
   266→    if(this._mode_callback!=='') E.Modes.on_mode_combat_end(this._mode_callback, 'fail', this.stage);
   267→    if(this.mode==='campaign'){
   268→      this._restart_pending=true;
   269→      this._spawn_t=this.RESPAWN_DELAY;
   270→    } else {
   271→      this.active=false;
   272→    }
   273→  },
   274→  _next_after_win: function(){
   275→    if(this.mode==='campaign'){ this.stage=E.Prog.current_stage; this._begin(); } // [HTML-5]
   276→    else { this.active=false; E.Modes.request_next(this.mode); }
   277→  }
   278→};

   280→/* ---------- SAVE (checksum SHA-256, backup, migração, anticheat) ---------- */
   281→E.Save = {
   282→  SAVE_VERSION: 1,
   283→  SALT: 'EIDRYN_VELUN_CICLO_DO_ECLIPSE_SALT',
   284→  PASS: 'EIDRYN_SeloDoCrepusculo_EIDRYN_VELUN_CICLO_DO_ECLIPSE_SALT',
   285→  FLUSH_DEBOUNCE_S: 3,
   286→  KEY: 'eidryn_save_v1',
   287→  BAK_KEY: 'eidryn_save_bak_v1',
   288→  _dirty: false, _timer: 0, _boot_done: false, _violations: [],
   289→  _store: {
   290→    get: function(k){ try{ return globalThis.localStorage.getItem(k); }catch(e){ return null; } },
   291→    set: function(k,v){ try{ globalThis.localStorage.setItem(k,v); }catch(e){ console.error('save storage cheio?', e); } },
   292→    del: function(k){ try{ globalThis.localStorage.removeItem(k); }catch(e){} }
   293→  },
   294→  /* obfuscação leve (paridade do AES do Godot — camada de ofuscação local) */
   295→  _xor: function(text){
   296→    var out='';
   297→    for(var i=0;i<text.length;i++) out+=String.fromCharCode(text.charCodeAt(i) ^ this.PASS.charCodeAt(i%this.PASS.length));
   298→    return out;
   299→  },
   300→  boot_load: function(){
   301→    var d=this._read(this.KEY);
   302→    if(!d){
   303→      var bak=this._read(this.BAK_KEY);
   304→      if(bak){ d=bak; E.BUS.toast_msg(E.DM.tr('save_corrupt'), '#e8a33a'); }
   305→    }
   306→    if(!d) d={};
   307→    this._apply_migrations(d);
   308→    this._distribute(d);
   309→    this._boot_done=true;
   310→    this._validate_all();
   311→    E.BUS.emit('save_flushed');
   312→  },
   313→  _read: function(key){
   314→    var raw=this._store.get(key);
   315→    if(!raw) return null;
   316→    try{
   317→      var text=this._xor(atob_compat(raw));
   318→      var payload=JSON.parse(text);
   319→      var data_json=payload.data_json||'';
   320→      if(data_json==='' || payload.checksum!==E.U.sha256(data_json)) return null;
   321→      return JSON.parse(data_json);
   322→    }catch(e){ return null; }
   323→  },
   324→  flush: function(){
   325→    if(!this._boot_done) return;
   326→    var data=this.collect();
   327→    var data_json=JSON.stringify(data);
   328→    var payload={version:this.SAVE_VERSION, checksum:E.U.sha256(data_json), data_json:data_json};
   329→    var text=JSON.stringify(payload);
   330→    var prev=this._store.get(this.KEY);
   331→    if(prev) this._store.set(this.BAK_KEY, prev);
   332→    this._store.set(this.KEY, btoa_compat(this._xor(text)));
   333→    this._dirty=false;
   334→    E.TimeM.mark_seen();
   335→    E.BUS.emit('save_flushed');
   336→  },
   337→  collect: function(){
   338→    return {
   339→      version:this.SAVE_VERSION,
   340→      time:E.TimeM.save_state(),
   341→      economy:E.Eco.save_state(),
   342→      character:E.Char.save_state(),
   343→      inventory:E.Inv.save_state(),
   344→      skills:E.Skill.save_state(),
   345→      pets:E.Pet.save_state(),
   346→      progression:E.Prog.save_state(),
   347→      modes:E.Modes.save_state(),
   348→      ascension:E.Asc.save_state(),
   349→      gacha:E.Gacha.save_state(),
   350→      retention:E.Ret.save_state(),
   351→      audio:E.Audio.save_state(),
   352→      system:{language:E.DM.language, game_version:E.DM.GAME_VERSION}
   353→    };
   354→  },
   355→  _distribute: function(d){
   356→    E.DM.language=((d.system||{}).language)||'ptbr';
   357→    E.TimeM.load_state(d.time||{});
   358→    E.Eco.load_state(d.economy||{});
   359→    E.Char.load_state(d.character||{});
   360→    E.Inv.load_state(d.inventory||{});
   361→    E.Skill.load_state(d.skills||{});
   362→    E.Pet.load_state(d.pets||{});
   363→    E.Prog.load_state(d.progression||{});
   364→    E.Modes.load_state(d.modes||{});
   365→    E.Asc.load_state(d.ascension||{});
   366→    E.Gacha.load_state(d.gacha||{});
   367→    E.Ret.load_state(d.retention||{});
   368→    E.Audio.load_state(d.audio||{});
   369→  },
   370→  _apply_migrations: function(d){
   371→    var v=d.version||0;
   372→    while(v<this.SAVE_VERSION){ v++; } // v0 → v1: estrutura idêntica
   373→    d.version=this.SAVE_VERSION;
   374→  },
   375→  mark_dirty: function(){ this._dirty=true; },
   376→  process: function(delta){
   377→    if(!this._boot_done) return;
   378→    if(this._dirty){
   379→      this._timer+=delta;
   380→      if(this._timer>=this.FLUSH_DEBOUNCE_S){ this._timer=0; this.flush(); }
   381→    } else this._timer=0;
   382→  },
   383→  export_save: function(){
   384→    var data=this.collect();
   385→    var data_json=JSON.stringify(data);
   386→    return btoa_compat(unescape(encodeURIComponent(JSON.stringify({version:this.SAVE_VERSION, checksum:E.U.sha256(data_json), data_json:data_json}))));
   387→  },
   388→  import_save: function(b64){
   389→    try{
   390→      var text=decodeURIComponent(escape(atob_compat(b64.trim())));
   391→      var payload=JSON.parse(text);
   392→      if(payload.checksum!==E.U.sha256(payload.data_json)) return false;
   393→      var d=JSON.parse(payload.data_json);
   394→      this._apply_migrations(d);
   395→      this._distribute(d);
   396→      this._validate_all();
   397→      E.Char.recalc();
   398→      this.flush();
   399→      return true;
   400→    }catch(e){ return false; }
   401→  },
   402→  _validate_all: function(){
   403→    this._violations=[];
   404→    var cur=E.Eco.currencies, caps=E.DM.cfg_currencies.caps;
   405→    for(var id in cur){
   406→      if(Number(cur[id])<0){ cur[id]=0; this._violations.push('moeda negativa: '+id); }
   407→      if(caps[id]!=null && Number(cur[id])>Number(caps[id])){ cur[id]=Number(caps[id]); this._violations.push('moeda acima do cap: '+id); }
   408→    }
   409→    var cap=E.DM.cfg_attributes.level.cap;

   537→  _noiseBuf: function(secs){
