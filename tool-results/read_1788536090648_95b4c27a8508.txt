   271→  /* ================= FSM DO HERÓI ================= */
   272→  _heroPose: function(p, force){
   273→    var def = this.POSES[p], frames = this.heroFrames[p];
   274→    if (!def || !frames || !frames.length) return;
   275→    var curP = this.heroAnim.pose, cur = this.POSES[curP];
   276→    if (!force && cur && def.prio < cur.prio) {
   277→      var curFr = this.heroFrames[curP] || [];
   278→      var curDone = cur.loop ? false : (this.heroAnim.t * cur.fps >= curFr.length);
   279→      if (!curDone) return; // não interrompe postura mais prioritária
   280→    }
   281→    if (!force && p === curP && this.heroAnim.t < 0.05) return;
   282→    this.heroAnim.pose = p; this.heroAnim.f = 0; this.heroAnim.t = 0;
   283→  },
   284→  _heroTick: function(delta){
   285→    var a = this.heroAnim, def = this.POSES[a.pose], fr = this.heroFrames[a.pose];
   286→    if (!def || !fr || !fr.length) { a.pose = 'idle'; a.f = 0; a.t = 0; return; }
   287→    a.t += delta;
   288→    var idx = Math.floor(a.t * def.fps);
   289→    if (def.loop) { a.f = idx % fr.length; }
   290→    else if (idx >= fr.length) {
   291→      if (def.hold) { a.f = fr.length - 1; } // segura até evento (spawn/respawn)
   292→      else {
   293→        var nx = def.next && this.heroFrames[def.next] ? def.next : 'idle';
   294→        a.pose = nx; a.t = 0; a.f = 0;
   295→      }
   296→    } else a.f = idx;
   297→  },

   303→    if (w) this.heroWhiteCache[key] = w;
   304→    return w;

   372→  _shiftHex: function(hex, S, k, br){
   373→    var r = parseInt(hex.substr(1, 2), 16), g = parseInt(hex.substr(3, 2), 16), b = parseInt(hex.substr(5, 2), 16);
   374→    var lum = (r * 0.3 + g * 0.59 + b * 0.11) / 255;
   375→    var t = S.tint, w = S.amt * k;
   376→    r = (r * br) * (1 - w * 0.7) + t[0] * w * 0.7 * (0.35 + lum * 0.65);
   377→    g = (g * br) * (1 - w * 0.7) + t[1] * w * 0.7 * (0.35 + lum * 0.65);
   378→    b = (b * br) * (1 - w * 0.7) + t[2] * w * 0.7 * (0.35 + lum * 0.65);
   379→    return 'rgb(' + (r > 255 ? 255 : r | 0) + ',' + (g > 255 ? 255 : g | 0) + ',' + (b > 255 ? 255 : b | 0) + ')';

   391→        if (!px[i + 3]) continue;

   395→        r *= br; g *= br; b *= br;
   396→        var w = amt * (0.3 + lum / 255 * 0.7);
   397→        r = r * (1 - w) + t[0] * w * 1.06;
   398→        g = g * (1 - w) + t[1] * w * 1.06;

   528→  _rainTick: function(delta){
   529→    var W = this.WEATHERS[this.weather] || this.WEATHERS.limpo;
   530→    var def = W.rain || W.snow;
   531→    var target = def ? Math.round(def.n * (0.35 + 0.65 * this.weatherBlend)) : 0;
   532→    // repovoamento/esvaziamento gradual (sem pop)
   533→    if (this.rainOn < target) this.rainOn = Math.min(target, this.rainOn + 5);

   536→    var alive = 0;
   537→    for (var i = 0; i < this.rain.length; i++) {
   538→      var p = this.rain[i];
   539→      if (!p.on) {
   540→        if (alive < this.rainOn) { this._resetDrop(p, true); alive++; }
   541→        continue;
   542→      }
   543→      var wrapped = false;
   544→      if (isSnow || isAsh) {
   545→        // neve / cinzas: deriva suave com vento
   546→        p.y += p.sp * delta * (isAsh ? 0.35 : 0.45);
   547→        p.x += (Math.sin(this.time * 1.2 + p.ph) * 16 + this.wind * 46) * delta;
   548→        if (p.y > this.GROUND_Y + 6) wrapped = true;
   549→        if (p.x > this.W + 8) { p.x = -8; }
   550→      } else {
   551→        p.y += p.sp * delta;
   552→        p.x += this.wind * 175 * delta;
   553→        if (p.y > this.GROUND_Y - 4 + p.gy) {
   554→          if (p.front && Math.random() < 0.34) {
   555→            this._splash(p.x, this.GROUND_Y + 2 + p.gy);
   556→            this._puddleRippleAt(p.x, this.GROUND_Y + 2 + p.gy); // [ART-11] ondulação na poça
   557→          }
   558→          wrapped = true;
   559→        }
   560→        if (p.x > this.W + 12) p.x = -10;
   561→      }
   562→      if (wrapped) {
   563→        alive++;
   564→        if (alive > this.rainOn) { p.on = false; continue; }
   565→        this._resetDrop(p, true);
   566→      } else alive++;
   567→    }
   568→    // splashes do solo
   569→    for (var s = 0; s < this.splashes.length; s++) {
   570→      var sp = this.splashes[s];
   571→      if (!sp.on) continue;
   572→      sp.t += delta;
   573→      if (sp.t >= sp.life) { sp.on = false; continue; }
   574→    }
